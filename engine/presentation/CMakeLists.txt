cmake_minimum_required(VERSION 3.16)
project(P5CastPresentation LANGUAGES C)

find_package(SDL3 3.2 CONFIG REQUIRED)
add_library(P5CastPresentation SHARED Presenter.c)
target_compile_features(P5CastPresentation PRIVATE c_std_11)
target_include_directories(P5CastPresentation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(P5CastPresentation PRIVATE SDL3::SDL3)
if(WIN32)
    target_link_libraries(P5CastPresentation PRIVATE user32)
endif()
set_target_properties(P5CastPresentation PROPERTIES OUTPUT_NAME P5Cast.Presentation)

option(BUILD_TESTING "Build the presentation driver probe" OFF)
if(BUILD_TESTING)
    enable_testing()
    add_executable(P5CastPresenterProbe ../../tests/PresenterProbe.c)
    target_compile_features(P5CastPresenterProbe PRIVATE c_std_11)
    target_link_libraries(P5CastPresenterProbe PRIVATE SDL3::SDL3 P5CastPresentation)
    if(WIN32)
        set(drivers direct3d12 vulkan opengl)
    elseif(APPLE)
        set(drivers metal)
    elseif(ANDROID)
        set(drivers vulkan opengles2)
    elseif(UNIX)
        set(drivers vulkan opengl)
    endif()
    foreach(driver IN LISTS drivers)
        add_test(NAME "present-${driver}" COMMAND P5CastPresenterProbe "${driver}")
    endforeach()
endif()
