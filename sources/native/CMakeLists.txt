cmake_minimum_required(VERSION 3.20)
project(CadrivectDirect LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
foreach(option TESTS CLI GUI BOREALIS STEAMDECK_NATIVE SETSU SPEEX FFMPEG_DECODER PI_DECODER STEAM_SHORTCUT)
    set(CHIAKI_ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CHIAKI_USE_SYSTEM_CURL ON CACHE BOOL "" FORCE)
set(CHIAKI_USE_SYSTEM_JERASURE OFF CACHE BOOL "" FORCE)
set(CHIAKI_USE_SYSTEM_NANOPB OFF CACHE BOOL "" FORCE)
set(CHIAKI_LIB_ENABLE_OPUS ON CACHE BOOL "" FORCE)
add_subdirectory("${CHIAKI_SOURCE}" chiaki EXCLUDE_FROM_ALL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_executable(CadrivectDirect Direct.c Video.c Audio.c Gamepad.c Register.c Rest.c)
target_compile_definitions(CadrivectDirect PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CadrivectDirect PRIVATE chiaki-lib PkgConfig::SDL2)
target_compile_options(CadrivectDirect PRIVATE -Wall -Wextra)
add_executable(CadrivectDirectVerify EXCLUDE_FROM_ALL Verify.c Video.c Audio.c Gamepad.c Rest.c)
target_compile_definitions(CadrivectDirectVerify PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CadrivectDirectVerify PRIVATE chiaki-lib PkgConfig::SDL2)
